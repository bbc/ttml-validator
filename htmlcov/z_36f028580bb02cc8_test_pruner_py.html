38 'clipEnd',

60 def test_pruner(self): 

61 pruner = Pruner( 

62 no_prune_namespaces=self.recognised_namespaces, 

63 no_prune_no_namespace_attributes=self.known_no_ns_attributes) 

64 

65 input_xml = """<?xml version="1.0" encoding="UTF-8"?> 

66<tt xmlns="http://www.w3.org/ns/ttml" 

67 xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance" 

68 xmlns:tta="http://www.w3.org/ns/ttml#audio" 

69 xmlns:ttm="http://www.w3.org/ns/ttml#metadata" 

70 xmlns:ttp="http://www.w3.org/ns/ttml#parameter" 

71 xmlns:daptm="http://www.w3.org/ns/ttml/profile/dapt#metadata" 

72 xmlns:ebuttm="urn:ebu:tt:metadata" 

73 ttp:contentProfiles="http://www.w3.org/ns/ttml/profile/dapt1.0/content" 

74 daptm:scriptRepresents="audio" 

75 daptm:scriptType="originalTranscript" 

76 xml:lang="en"> 

77 <head> 

78 <metadata xmlns:otherns="urn:some:other:namespace"> 

79 <ttm:agent xmlns:nttm="http://www.netflix.com/ns/ttml#metadata" 

80 nttm:voice="en-US-Wavenet-B" foo="bar" otherns:test="x" type="person" xml:id="actor_A" id="huh"> 

81 <ttm:name type="full">Matthias Schoenaerts</ttm:name> 

82 </ttm:agent> 

83 <ttm:agent type="character" xml:id="character_2"> 

84 <ttm:name type="alias">BOOKER</ttm:name> 

85 <ttm:actor agent="actor_A"/> 

86 </ttm:agent> 

87 <otherns:x/> 

88 <otherns:y/> 

89 </metadata> 

90 </head> 

91 <body> 

92 <div xml:id="se1" begin="3s" end="10s" ttm:agent="character_2" daptm:represents="audio.dialogue" daptm:onScreen="ON"> 

93 <ttm:desc daptm:descType="scene">high mountain valley</ttm:desc> 

94 <metadata></metadata> 

95 <p daptm:langSrc="en"><span>Look at this beautiful valley.</span></p> 

96 </div> 

97 </body> 

98</tt> 

99""" # noqa: E501 

100 input_etree = ElementTree.fromstring(input_xml) 

101 vr = ValidationLogger() 

102 context = {} 

103 valid = pruner.run( 

104 input=input_etree, 

105 context=context, 

106 validation_results=vr) 

107 self.assertListEqual(vr, [ 

108 ValidationResult( 

109 status=INFO, 

110 location='Document', 

111 message='Pruned 0 elements and 1 attributes ' 

112 '("voice" 1 time) in namespace ' 

113 '"http://www.netflix.com/ns/ttml#metadata"', 

114 code=ValidationCode.xml_prune 

115 ), 

116 ValidationResult( 

117 status=INFO, 

118 location='Document', 

119 message='Pruned 0 elements and 2 attributes ' 

120 '("foo" 1 time, "id" 1 time) in namespace ""', 

121 code=ValidationCode.xml_prune 

122 ), 

123 ValidationResult( 

124 status=INFO, 

125 location='Document', 

126 message='Pruned 2 elements ("x" 1 time, "y" 1 time) and ' 

127 '1 attributes ("test" 1 time) in namespace ' 

128 '"urn:some:other:namespace"', 

129 code=ValidationCode.xml_prune 

130 ) 

131 ]) 

132 self.assertTrue(valid) 

133 # To get the output namespaces to look the way we want, 

134 # register them explicitly 

135 ElementTree.register_namespace('', 'http://www.w3.org/ns/ttml') 

136 ElementTree.register_namespace('ttp', 'http://www.w3.org/ns/ttml#parameter') 

137 ElementTree.register_namespace('ttm', 'http://www.w3.org/ns/ttml#metadata') 

138 ElementTree.register_namespace('daptm', 'http://www.w3.org/ns/ttml/profile/dapt#metadata') 

139 ElementTree.register_namespace('ebuttm', 'urn:ebu:tt:metadata') 

140 

141 output_str = ElementTree \ 

142 .tostring(input_etree, 

143 encoding='UTF-8', 

144 xml_declaration=True) \ 

145 .decode('utf-8') 

146 # ElementTree.tostring() actually makes bytes! 

147 

148 # The format of output that ElementTree makes is a bit weird, but short 

149 # of reparsing it and processing it as XML again, this is probably the 

150 # easiest thing to do. It may be a bit brittle... 

151 expected_output_str = """<?xml version='1.0' encoding='UTF-8'?> 

152<tt xmlns="http://www.w3.org/ns/ttml" xmlns:daptm="http://www.w3.org/ns/ttml/profile/dapt#metadata" xmlns:ttm="http://www.w3.org/ns/ttml#metadata" xmlns:ttp="http://www.w3.org/ns/ttml#parameter" ttp:contentProfiles="http://www.w3.org/ns/ttml/profile/dapt1.0/content" daptm:scriptRepresents="audio" daptm:scriptType="originalTranscript" xml:lang="en"> 

153 <head> 

154 <metadata> 

155 <ttm:agent type="person" xml:id="actor_A"> 

156 <ttm:name type="full">Matthias Schoenaerts</ttm:name> 

157 </ttm:agent> 

158 <ttm:agent type="character" xml:id="character_2"> 

159 <ttm:name type="alias">BOOKER</ttm:name> 

160 <ttm:actor agent="actor_A" /> 

161 </ttm:agent> 

162 </metadata> 

163 </head> 

164 <body> 

165 <div xml:id="se1" begin="3s" end="10s" ttm:agent="character_2" daptm:represents="audio.dialogue" daptm:onScreen="ON"> 

166 <ttm:desc daptm:descType="scene">high mountain valley</ttm:desc> 

167 <metadata /> 

168 <p daptm:langSrc="en"><span>Look at this beautiful valley.</span></p> 

169 </div> 

170 </body> 

171</tt>""" # noqa: E501 

172 self.assertEqual(output_str, expected_output_str)
